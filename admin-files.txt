@router.get("/users", response_model=List[UserResponse])
async def admin_users(current_admin: User = Depends(require_admin), db: AsyncSession = Depends(get_db)):
    res = await db.execute(select(User))
    return res.scalars().all()

@router.post("/users", response_model=UserResponse, status_code=201)
async def create_admin_user(
    user_in: AdminUserCreate,
    current_admin: User = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    phone = normalize_ke_phone(user_in.phone)
    existing = await db.execute(select(User).where(User.phone == phone))
    if existing.scalar_one_or_none():
        raise HTTPException(status_code=409, detail="Phone already registered")
    user = User(
        name=user_in.name,
        phone=phone,
        password_hash=get_password_hash(user_in.password),
        role="admin",
    )
    db.add(user)
    await db.commit()
    await db.refresh(user)
    return user

@router.get("/farms", response_model=List[FarmResponse])
async def admin_farms(
    include_deleted: bool = False,
    current_admin: User = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    query = select(Farm)
    if not include_deleted:
        query = query.where(Farm.deleted_at.is_(None))
    res = await db.execute(query)
    return res.scalars().all()

@router.patch("/model-versions/{id}", response_model=ModelVersionResponse)
async def update_model_version(id: uuid.UUID, update_data: ModelVersionUpdate, current_admin: User = Depends(require_admin), db: AsyncSession = Depends(get_db)):
    res = await db.execute(select(ModelVersion).where(ModelVersion.id == id))
    mv = res.scalar_one_or_none()
    if not mv: raise HTTPException(404, "Not found")
    if update_data.is_active:
        active_versions = await db.execute(
            select(ModelVersion).where(
                ModelVersion.model_name == mv.model_name,
                ModelVersion.is_active.is_(True),
                ModelVersion.id != mv.id,
            )
        )
        for active_version in active_versions.scalars():
            active_version.is_active = False
    mv.is_active = update_data.is_active
    await db.commit()
    await db.refresh(mv)
    return mv

"""Send an SMS through Africa's Talking. Never raises: returns True only when the first recipient reports Success (101).

The secure connection uses Python's built-in client and is capped at TLS 1.2. On some networks and security tools the
newer TLS 1.3 handshake, and the requests library's own connection settings, fail with WRONG_VERSION_NUMBER while a
plain TLS 1.2 connection works. Africa's Talking supports TLS 1.2."""
import json
import os
import ssl
import urllib.error
import urllib.parse
import urllib.request

TIMEOUT_SECONDS = 10


def _endpoint(username: str) -> str:
    host = "api.sandbox.africastalking.com" if username == "sandbox" else "api.africastalking.com"
    return f"https://{host}/version1/messaging"


def _tls12_context() -> ssl.SSLContext:
    context = ssl.create_default_context()
    context.maximum_version = ssl.TLSVersion.TLSv1_2
    return context


def send_sms(phone_number: str, message: str) -> bool:
    try:
        username = os.getenv("AFRICASTALKING_USERNAME", "sandbox")
        api_key = os.getenv("AFRICASTALKING_API_KEY", "dummy_key")
        body = urllib.parse.urlencode({"username": username, "to": phone_number, "message": message}).encode()
        request = urllib.request.Request(
            _endpoint(username),
            data=body,
            headers={"apiKey": api_key, "Accept": "application/json",
                     "Content-Type": "application/x-www-form-urlencoded"},
            method="POST",
        )
        with urllib.request.urlopen(request, timeout=TIMEOUT_SECONDS, context=_tls12_context()) as response:
            payload = json.loads(response.read().decode("utf-8"))
        if not isinstance(payload, dict):
            return False
        recipients = payload.get("SMSMessageData", {}).get("Recipients", [])
        if not recipients or not isinstance(recipients, list):
            return False
        first = recipients[0]
        if not isinstance(first, dict):
            return False
        return first.get("status") == "Success" and first.get("statusCode") == 101
    except Exception as e:
        print(f"Failed to send SMS to {phone_number}: {str(e)}")
        return False

    none: { tone: 'neutral', label: 'Not sure' },
}

// Shows "Disease Confirmed" only when the model is confident. A weaker result gets a softer badge.
function DiseaseBadge({ band, label, level }) {
    const isDisease = Boolean(label) && label.toLowerCase() !== 'healthy'
    const style = isDisease ? BAND_BADGES[band] : null
    if (!style) return <SeverityBadge level={level} />
    return <Badge tone={style.tone}>{style.label}</Badge>
}

function PlotReadingsCard({ farmId }) {

        ['Humidity', `${reading.humidity}%`],

            {label && (

import { useState } from 'react'
import { focusRing } from './styles'

function compare(a, b) {
  if (a == null && b == null) return 0
  if (a == null) return 1
  if (b == null) return -1
  if (typeof a === 'number' && typeof b === 'number') return a - b
  return String(a).localeCompare(String(b), undefined, { numeric: true, sensitivity: 'base' })
}

function SortArrow({ dir }) {
  return (
    <svg aria-hidden="true" viewBox="0 0 12 12" className={`h-3 w-3 shrink-0 ${dir ? 'opacity-100' : 'opacity-40'}`} fill="currentColor">
      {dir !== 'desc' && <path d="M6 1.5 9.5 5h-7z" />}
      {dir !== 'asc' && <path d="M6 10.5 2.5 7h7z" />}
    </svg>
  )
}

// Dense admin table: header stays visible while the body scrolls, sortable columns announce
// their state with aria-sort, numbers right-align in tabular figures.
// columns: [{ key, label, render(row), sortValue?(row), align?: 'right', className? }]
export default function DataTable({ caption, columns, rows, rowKey, initialSort, empty, maxHeight = 'max-h-[70vh]' }) {
  const [sort, setSort] = useState(initialSort ?? null)
  const sortColumn = columns.find((column) => column.key === sort?.key)
  const sorted = sortColumn?.sortValue
    ? [...rows].sort((a, b) => compare(sortColumn.sortValue(a), sortColumn.sortValue(b)) * (sort.dir === 'desc' ? -1 : 1))
    : rows

  const toggle = (key) =>
    setSort((current) => (current?.key === key ? { key, dir: current.dir === 'asc' ? 'desc' : 'asc' } : { key, dir: 'asc' }))

  if (rows.length === 0) {
    return <p className="m-0 rounded-control bg-app-bg p-5 text-center text-small text-ink-secondary">{empty}</p>
  }

  return (
    <div className={`overflow-auto rounded-control border border-border ${maxHeight}`}>
      <table className="w-full min-w-[720px] border-collapse text-left text-small">
        <caption className="sr-only">{caption}</caption>
        <thead>
          <tr>
            {columns.map((column) => {
              const dir = sort?.key === column.key ? sort.dir : null
              return (
                <th
                  key={column.key}
                  scope="col"
                  aria-sort={column.sortValue ? (dir === 'asc' ? 'ascending' : dir === 'desc' ? 'descending' : 'none') : undefined}
                  className={`sticky top-0 z-10 border-b border-border bg-app-bg px-4 text-caption font-bold text-ink-secondary ${column.align === 'right' ? 'text-right' : ''}`}
                >
                  {column.sortValue ? (
                    <button
                      type="button"
                      onClick={() => toggle(column.key)}
                      className={`inline-flex min-h-11 items-center gap-1.5 rounded-control bg-transparent px-0 text-caption font-bold text-ink-secondary hover:text-ink ${focusRing} ${column.align === 'right' ? 'flex-row-reverse' : ''}`}
                    >
                      {column.label}
                      <SortArrow dir={dir} />
                    </button>
                  ) : (
                    <span className="inline-flex min-h-11 items-center">{column.label}</span>
                  )}
                </th>
              )
            })}
          </tr>
        </thead>
        <tbody className="divide-y divide-border bg-surface">
          {sorted.map((row) => (
            <tr key={rowKey(row)} className="hover:bg-app-bg">
              {columns.map((column) => (
                <td
                  key={column.key}
                  className={`px-4 py-3 align-top ${column.align === 'right' ? 'text-right tabular-nums' : ''} ${column.className ?? ''}`}
                >
                  {column.render(row)}
                </td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}
